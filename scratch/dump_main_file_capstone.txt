import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.metrics import confusion_matrix

class_names = train_data.class_names

y_true = np.concatenate([y.numpy() for x, y in validation_data])
y_pred_prob = model.predict(validation_data)
y_pred = np.argmax(y_pred_prob, axis=1)

cm = confusion_matrix(y_true, y_pred)

plt.figure(figsize=(16, 14))

sns.heatmap(
    cm,
    annot=True,
    fmt="d",
    cmap="Blues",
    xticklabels=class_names,
    yticklabels=class_names
)

plt.xlabel("Predicted Class")
plt.ylabel("Actual Class")
plt.title("Confusion Matrix Heatmap - ResNet50")
plt.xticks(rotation=90)
plt.yticks(rotation=0)

plt.tight_layout()

plt.savefig(
    "resnet50_confusion_matrix.png",
    dpi=300,
    bbox_inches="tight"
)

plt.show()

from transformers import AutoProcessor, Florence2ForConditionalGeneration

florence_processor = AutoProcessor.from_pretrained(
    "florence-community/Florence-2-base"
)

florence_model = Florence2ForConditionalGeneration.from_pretrained(
    "florence-community/Florence-2-base",
    torch_dtype=torch.float16,
    device_map="auto"
)

--- CELL 35 (code) ---
task_prompt = "<CAPTION>"

inputs = florence_processor(
    text=task_prompt,
    images=sample_pil,
    return_tensors="pt"
)

inputs = {
    k: v.to(florence_model.device)
    for k, v in inputs.items()
}

with torch.no_grad():
    generated_ids = florence_model.generate(
        **inputs,
        max_new_tokens=100,
        num_beams=3
    )

generated_text = florence_processor.batch_decode(
    generated_ids,
    skip_special_tokens=True
)[0]

print("Florence-2 Description:")
print(generated_text)
